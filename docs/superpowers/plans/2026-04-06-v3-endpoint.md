# V3 Endpoint Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add self-contained v3 typecheck/compile endpoints that accept multipart file uploads with caller-provided deps, tsconfig, and esbuild config.

**Architecture:** New v3 handlers parse multipart/form-data, resolve deps via a 3-tier cache (local disk -> S3 -> bun install), read compiler/esbuild options from uploaded tsconfig.json and package.json, then run the existing TypeScript-Go compiler and esbuild bundler against the blended filesystem.

**Tech Stack:** Go net/http, multipart, esbuild Go API, TypeScript-Go compiler, bun (runtime dep install), AWS S3 SDK, tar/gzip

**Spec:** `docs/superpowers/specs/2026-04-06-v3-endpoint-design.md`

```go
package main

import (
	"fmt"
	"io"
	"mime"
	"mime/multipart"
	"strings"
)

// parseV3Multipart reads a multipart/form-data request body and returns a map of path -> content.
// Validates required files are present and enforces limits.
func parseV3Multipart(body io.Reader, contentType string) (map[string][]byte, error) {
	mediaType, params, err := mime.ParseMediaType(contentType)
	if err != nil {
		return nil, fmt.Errorf("invalid content type: %w", err)
	}
	if mediaType != "multipart/form-data" {
		return nil, fmt.Errorf("expected multipart/form-data, got %s", mediaType)
	}

	boundary := params["boundary"]
	if boundary == "" {
		return nil, fmt.Errorf("missing boundary in content type")
	}

	reader := multipart.NewReader(body, boundary)
	files := make(map[string][]byte)
	var totalSize int64

	for {
		part, err := reader.NextPart()
		if err == io.EOF {
			break
		}
		if err != nil {
			return nil, fmt.Errorf("error reading multipart: %w", err)
		}

		path := part.FormName()
		if path == "" {
			continue
		}

		// Normalize path
		if !strings.HasPrefix(path, "/") {
			path = "/" + path
		}

		// Reject node_modules paths
		if strings.HasPrefix(path, "/node_modules/") || path == "/node_modules" {
			return nil, fmt.Errorf("cannot upload node_modules files: %s", path)
		}

		// Enforce file count limit (package.json and bun.lock count toward limit)
		if len(files) >= maxFilesPerRequest+2 {
			return nil, fmt.Errorf("too many files: max %d (plus package.json and bun.lock)", maxFilesPerRequest)
		}

		// Read file content with size limit
		data, err := io.ReadAll(io.LimitReader(part, maxFileSizeBytes+1))
		if err != nil {
			return nil, fmt.Errorf("error reading file %s: %w", path, err)
		}
		if int64(len(data)) > maxFileSizeBytes {
			return nil, fmt.Errorf("file %s exceeds max size of %d bytes", path, maxFileSizeBytes)
		}

		totalSize += int64(len(data))
		if totalSize > maxTotalSizeBytes {
			return nil, fmt.Errorf("total payload exceeds max size of %d bytes", maxTotalSizeBytes)
		}

		files[path] = data
	}

	// Validate required files
	if _, ok := files["/package.json"]; !ok {
		return nil, fmt.Errorf("missing required file: /package.json")
	}
	if _, ok := files["/bun.lock"]; !ok {
		return nil, fmt.Errorf("missing required file: /bun.lock")
	}

	return files, nil
}
```

```go
import (
	"encoding/json"

	"github.com/evanw/esbuild/pkg/api"
)

// v3PackageJSON represents the parsed package.json for v3 requests
type v3PackageJSON struct {
	Dependencies    map[string]string `json:"dependencies"`
	DevDependencies map[string]string `json:"devDependencies"`
	Esbuild         v3EsbuildConfig   `json:"esbuild"`
	Main            string            `json:"main"`
}

// v3EsbuildConfig represents the esbuild configuration from package.json
type v3EsbuildConfig struct {
	Bundle            *bool    `json:"bundle"`
	External          []string `json:"external"`
	Format            string   `json:"format"`
	Minify            *bool    `json:"minify"`
	MinifyIdentifiers *bool    `json:"minifyIdentifiers"`
	MinifySyntax      *bool    `json:"minifySyntax"`
	MinifyWhitespace  *bool    `json:"minifyWhitespace"`
	Platform          string   `json:"platform"`
	Target            string   `json:"target"`
}

func parsePackageJSON(raw []byte) (*v3PackageJSON, error) {
	var pkg v3PackageJSON
	if err := json.Unmarshal(raw, &pkg); err != nil {
		return nil, fmt.Errorf("invalid package.json: %w", err)
	}
	if pkg.Main == "" {
		return nil, fmt.Errorf("package.json missing required 'main' field")
	}
	return &pkg, nil
}

// esbuildOptions converts the parsed config into esbuild Go API BuildOptions fields.
// Returns only the fields that v3 controls; caller adds entry points, plugins, write mode, etc.
func (pkg *v3PackageJSON) esbuildOptions() api.BuildOptions {
	opts := api.BuildOptions{
		Bundle:            true,
		External:          []string{"*"},
		Format:            api.FormatCommonJS,
		MinifyIdentifiers: false,
		MinifySyntax:      true,
		MinifyWhitespace:  true,
		Platform:          api.PlatformBrowser,
		Target:            api.ES2022,
		Write:             false,
	}

	cfg := pkg.Esbuild

	if cfg.Bundle != nil {
		opts.Bundle = *cfg.Bundle
	}
	if len(cfg.External) > 0 {
		opts.External = cfg.External
	}

	switch cfg.Format {
	case "cjs":
		opts.Format = api.FormatCommonJS
	case "esm":
		opts.Format = api.FormatESModule
	case "iife":
		opts.Format = api.FormatIIFE
	}

	if cfg.Minify != nil {
		opts.MinifyIdentifiers = *cfg.Minify
		opts.MinifySyntax = *cfg.Minify
		opts.MinifyWhitespace = *cfg.Minify
	}
	if cfg.MinifyIdentifiers != nil {
		opts.MinifyIdentifiers = *cfg.MinifyIdentifiers
	}
	if cfg.MinifySyntax != nil {
		opts.MinifySyntax = *cfg.MinifySyntax
	}
	if cfg.MinifyWhitespace != nil {
		opts.MinifyWhitespace = *cfg.MinifyWhitespace
	}

	switch cfg.Platform {
	case "browser":
		opts.Platform = api.PlatformBrowser
	case "node":
		opts.Platform = api.PlatformNode
	case "neutral":
		opts.Platform = api.PlatformNeutral
	}

	switch cfg.Target {
	case "es2020":
		opts.Target = api.ES2020
	case "es2021":
		opts.Target = api.ES2021
	case "es2022":
		opts.Target = api.ES2022
	case "es2023":
		opts.Target = api.ES2023
	case "esnext":
		opts.Target = api.ESNext
	}

	return opts
}
```

```go
	"path/filepath"
	"sync"
	"time"
)

// In-flight install deduplication
var (
	depInstallMu     sync.Mutex
	depInstallFlights = make(map[string]*depInstallResult)
)

type depInstallResult struct {
	done chan struct{}
	err  error
	path string
}

// hashBunLock returns the SHA256 hex digest of the bun.lock contents.
func hashBunLock(lockContent []byte) string {
	h := sha256.Sum256(lockContent)
	return hex.EncodeToString(h[:])
}

// resolveDeps returns the path to a directory containing node_modules for the given lockfile.
// Checks local disk first, then S3, then runs bun install.
func resolveDeps(ctx context.Context, lockContent []byte, packageJSON []byte) (string, error) {
	hash := hashBunLock(lockContent)
	depDir := filepath.Join(diskCachePath, "deps", hash)

	// 1. Local disk check
	nmDir := filepath.Join(depDir, "node_modules")
	if info, err := os.Stat(nmDir); err == nil && info.IsDir() {
		log.Printf("[DEPS] Local cache hit for %s", hash[:12])
		depCacheLookups.WithLabelValues("local_hit").Inc()
		// Touch the directory to update mtime for LRU
		now := time.Now()
		os.Chtimes(depDir, now, now)
		return depDir, nil
	}

	// 2. Deduplicate concurrent installs for same hash
	depInstallMu.Lock()
	if flight, ok := depInstallFlights[hash]; ok {
		depInstallMu.Unlock()
		log.Printf("[DEPS] Waiting on in-flight install for %s", hash[:12])
		<-flight.done
		return flight.path, flight.err
	}

	flight := &depInstallResult{done: make(chan struct{})}
	depInstallFlights[hash] = flight
	depInstallMu.Unlock()

	// When done, broadcast to all waiters and clean up
	defer func() {
		close(flight.done)
		depInstallMu.Lock()
		delete(depInstallFlights, hash)
		depInstallMu.Unlock()
	}()

	// 3. S3 check
	path, err := resolveDepsFromS3(ctx, hash, depDir)
	if err == nil {
		flight.path = path
		return path, nil
	}

	// 4. bun install
	path, err = installDeps(ctx, hash, depDir, lockContent, packageJSON)
	flight.path = path
	flight.err = err
	return path, err
}
```

```go
import (
	"archive/tar"
	"bytes"
	"compress/gzip"
	"fmt"
	"io"
	"os/exec"

	"github.com/aws/aws-sdk-go-v2/aws"
	"github.com/aws/aws-sdk-go-v2/service/s3"

// resolveDepsFromS3 downloads and extracts a cached dep tarball from S3.
func resolveDepsFromS3(ctx context.Context, hash string, depDir string) (string, error) {
	key := "deps/" + hash + ".tar.gz"

	result, err := s3Client.GetObject(ctx, &s3.GetObjectInput{
		Bucket: aws.String(s3Bucket),
		Key:    aws.String(key),
	})
	if err != nil {
		return "", fmt.Errorf("S3 cache miss for %s: %w", hash[:12], err)
	}
	defer result.Body.Close()

	log.Printf("[DEPS] S3 cache hit for %s, extracting", hash[:12])
	depCacheLookups.WithLabelValues("s3_hit").Inc()

	// Extract tarball to depDir
	if err := os.MkdirAll(depDir, 0755); err != nil {
		return "", fmt.Errorf("failed to create dep dir: %w", err)
	}

	gr, err := gzip.NewReader(result.Body)
	if err != nil {
		return "", fmt.Errorf("failed to open gzip: %w", err)
	}
	defer gr.Close()

	tr := tar.NewReader(gr)
	for {
		header, err := tr.Next()
		if err == io.EOF {
			break
		}
		if err != nil {
			return "", fmt.Errorf("tar read error: %w", err)
		}

		target := filepath.Join(depDir, header.Name)

		switch header.Typeflag {
		case tar.TypeDir:
			if err := os.MkdirAll(target, 0755); err != nil {
				return "", err
			}
		case tar.TypeReg:
			if err := os.MkdirAll(filepath.Dir(target), 0755); err != nil {
				return "", err
			}
			f, err := os.Create(target)
			if err != nil {
				return "", err
			}
			if _, err := io.Copy(f, tr); err != nil {
				f.Close()
				return "", err
			}
			f.Close()
		}
	}

	return depDir, nil
}

// installDeps runs bun install and caches the result locally and in S3.
func installDeps(ctx context.Context, hash string, depDir string, lockContent []byte, packageJSON []byte) (string, error) {
	log.Printf("[DEPS] Cache miss for %s, running bun install", hash[:12])
	depCacheLookups.WithLabelValues("miss").Inc()

	installStart := time.Now()
	defer func() {
		depInstallDuration.Observe(time.Since(installStart).Seconds())
	}()

	// Create temp directory for install
	tmpDir, err := os.MkdirTemp("", "bun-install-*")
	if err != nil {
		return "", fmt.Errorf("failed to create temp dir: %w", err)
	}

	// Clean up temp dir on error (on success we move it)
	success := false
	defer func() {
		if !success {
			os.RemoveAll(tmpDir)
		}
	}()

	// Write package.json and bun.lock
	if err := os.WriteFile(filepath.Join(tmpDir, "package.json"), packageJSON, 0644); err != nil {
		return "", fmt.Errorf("failed to write package.json: %w", err)
	}
	if err := os.WriteFile(filepath.Join(tmpDir, "bun.lock"), lockContent, 0644); err != nil {
		return "", fmt.Errorf("failed to write bun.lock: %w", err)
	}

	// Run bun install
	cmd := exec.CommandContext(ctx, "bun", "install", "--frozen-lockfile", "--ignore-scripts")
	cmd.Dir = tmpDir
	var stderr bytes.Buffer
	cmd.Stderr = &stderr

	if err := cmd.Run(); err != nil {
		return "", fmt.Errorf("bun install failed: %s", stderr.String())
	}

	log.Printf("[DEPS] bun install completed for %s in %v", hash[:12], time.Since(installStart))

	// Move to cache location
	if err := os.MkdirAll(filepath.Dir(depDir), 0755); err != nil {
		return "", fmt.Errorf("failed to create cache dir: %w", err)
	}
	if err := os.Rename(tmpDir, depDir); err != nil {
		// Rename may fail across filesystems; fall back to copy
		if err := copyDir(tmpDir, depDir); err != nil {
			return "", fmt.Errorf("failed to move deps to cache: %w", err)
		}
	}
	success = true

	// Upload to S3 in background (don't block the request)
	go uploadDepsToS3(context.Background(), hash, depDir)

	return depDir, nil
}

// uploadDepsToS3 tars and uploads the node_modules to S3.
func uploadDepsToS3(ctx context.Context, hash string, depDir string) {
	var buf bytes.Buffer
	gw := gzip.NewWriter(&buf)
	tw := tar.NewWriter(gw)

	nmDir := filepath.Join(depDir, "node_modules")
	err := filepath.Walk(nmDir, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}

		relPath, err := filepath.Rel(depDir, path)
		if err != nil {
			return err
		}

		header, err := tar.FileInfoHeader(info, "")
		if err != nil {
			return err
		}
		header.Name = relPath

		if err := tw.WriteHeader(header); err != nil {
			return err
		}

		if !info.IsDir() {
			f, err := os.Open(path)
			if err != nil {
				return err
			}
			defer f.Close()
			if _, err := io.Copy(tw, f); err != nil {
				return err
			}
		}
		return nil
	})

	if err != nil {
		log.Printf("[DEPS] Failed to create tarball for S3 upload: %v", err)
		return
	}

	tw.Close()
	gw.Close()

	key := "deps/" + hash + ".tar.gz"
	_, err = s3Client.PutObject(ctx, &s3.PutObjectInput{
		Bucket: aws.String(s3Bucket),
		Key:    aws.String(key),
		Body:   bytes.NewReader(buf.Bytes()),
	})
	if err != nil {
		log.Printf("[DEPS] Failed to upload deps to S3: %v", err)
		return
	}

	log.Printf("[DEPS] Uploaded %s to S3 (%d bytes)", key, buf.Len())
}

// copyDir recursively copies a directory.
func copyDir(src, dst string) error {
	return filepath.Walk(src, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		relPath, err := filepath.Rel(src, path)
		if err != nil {
			return err
		}
		targetPath := filepath.Join(dst, relPath)
		if info.IsDir() {
			return os.MkdirAll(targetPath, info.Mode())
		}
		data, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		return os.WriteFile(targetPath, data, info.Mode())
	})
}
```

```go
// parseTSConfig parses a tsconfig.json and returns compiler options.
// If tsconfigRaw is nil, returns default options.
func parseTSConfig(tsconfigRaw []byte) (*core.CompilerOptions, error) {
	if tsconfigRaw == nil {
		return defaultCompilerOptions(), nil
	}

	// Parse the tsconfig JSON to extract compilerOptions
	var tsconfig struct {
		CompilerOptions json.RawMessage `json:"compilerOptions"`
	}
	if err := json.Unmarshal(tsconfigRaw, &tsconfig); err != nil {
		return nil, err
	}

	// Start with defaults, then override with provided values
	opts := defaultCompilerOptions()

	if tsconfig.CompilerOptions == nil {
		return opts, nil
	}

	// Parse individual fields from compilerOptions
	var rawOpts map[string]json.RawMessage
	if err := json.Unmarshal(tsconfig.CompilerOptions, &rawOpts); err != nil {
		return nil, err

// typecheckV3 performs typechecking for a v3 request.
func typecheckV3(files map[string][]byte, tsconfigRaw []byte, lockContent []byte) TypecheckV2Response {
	typecheckStart := time.Now()
	defer func() {
		duration := time.Since(typecheckStart)
		typecheckDuration.Observe(duration.Seconds())
		log.Printf("[PERF] typecheckV3 total: %v (%d files)", duration, len(files))
	}()

	// Resolve deps
	hash := hashBunLock(lockContent)
	depDir := filepath.Join(diskCachePath, "deps", hash)

	fs := newDiskFSFromDeps(depDir)
	fs.hasUserFiles = true

	// Populate user files, collect entry points
	var fileNames []string
	for path, content := range files {
		normalized, err := normalizeAndValidatePath(path)
		if err != nil {
			return TypecheckV2Response{
				Errors: []DiagnosticErrorV2{{File: path, Message: err.Error()}},
			}
		}
		// Skip config files
		if normalized == "/package.json" || normalized == "/bun.lock" || normalized == "/tsconfig.json" {
			continue
		}
		fs.mu.Lock()
		fs.userFiles[normalized] = string(content)
		fs.mu.Unlock()
		if strings.HasSuffix(normalized, ".ts") || strings.HasSuffix(normalized, ".tsx") {
			fileNames = append(fileNames, normalized)
		}
	}

	if len(fileNames) == 0 {
		return TypecheckV2Response{
			Errors: []DiagnosticErrorV2{{Message: "No TypeScript files found to check"}},
		}
	}

	// Parse compiler options
	compilerOptions, err := parseTSConfig(tsconfigRaw)
	if err != nil {
		return TypecheckV2Response{
			Errors: []DiagnosticErrorV2{{Message: "invalid tsconfig.json: " + err.Error()}},
		}
	}

	wrappedFS := bundled.WrapFS(fs)

	parsedOptions := &core.ParsedOptions{
		CompilerOptions: compilerOptions,
		FileNames:       fileNames,
	}
	config := &tsoptions.ParsedCommandLine{
		ParsedConfig: parsedOptions,
	}
	extendedConfigCache := &tsc.ExtendedConfigCache{}
	host := compiler.NewCachedFSCompilerHost("/", wrappedFS, bundled.LibPath(), extendedConfigCache, nil)
	program := compiler.NewProgram(compiler.ProgramOptions{
		Config: config,
		Host:   host,
	})

	ctx := context.Background()

	diagnostics := program.GetSyntacticDiagnostics(ctx, nil)
	if len(diagnostics) == 0 {
		diagnostics = append(diagnostics, program.GetSemanticDiagnostics(ctx, nil)...)
	}

	if len(diagnostics) > 0 {
		errors := make([]DiagnosticErrorV2, 0, len(diagnostics))
		for _, diag := range diagnostics {
			diagErr := DiagnosticErrorV2{
				Message: diag.Localize(locale.Default),
			}
			if diag.File() != nil {
				diagErr.File = diag.File().FileName()
				if diag.Loc().Pos() >= 0 {
					line, col := calculateLineColumn(diag.File().Text(), diag.Loc().Pos())
					diagErr.Line = line + 1
					diagErr.Column = col + 1
				}
			}
			errors = append(errors, diagErr)
		}
		typecheckResults.WithLabelValues("error").Inc()
		return TypecheckV2Response{Errors: errors}
	}

	typecheckResults.WithLabelValues("success").Inc()
	return TypecheckV2Response{Pass: true}
}
```

```go
func compileV3(files map[string][]byte, pkg *v3PackageJSON, tsconfigRaw []byte, lockContent []byte) BuildV2Response {

		duration := time.Since(compileStart)
		compileDuration.Observe(duration.Seconds())
		log.Printf("[PERF] compileV3 total: %v (%d files)", duration, len(files))
	}()

	// Resolve deps
	hash := hashBunLock(lockContent)
	depDir := filepath.Join(diskCachePath, "deps", hash)

	fs := newDiskFSFromDeps(depDir)
	fs.hasUserFiles = true

	// Populate user files
	for path, content := range files {
		normalized, err := normalizeAndValidatePath(path)
		if err != nil {
			return BuildV2Response{
				Errors: []DiagnosticErrorV2{{File: path, Message: err.Error()}},
			}
		}
		if normalized == "/package.json" || normalized == "/bun.lock" || normalized == "/tsconfig.json" {
			continue
		}
		fs.mu.Lock()
		fs.userFiles[normalized] = string(content)
		fs.mu.Unlock()
	}

	// Resolve entry point from package.json main
	entryPoint := pkg.Main
	if !strings.HasPrefix(entryPoint, "/") {
		if strings.HasPrefix(entryPoint, "./") {
			entryPoint = entryPoint[1:] // "./src/index.ts" -> "/src/index.ts"
		} else {
			entryPoint = "/" + entryPoint
		}
	}

	// Verify entry point exists
	fs.mu.RLock()
	_, entryExists := fs.userFiles[entryPoint]
	fs.mu.RUnlock()
	if !entryExists {
		return BuildV2Response{
			Errors: []DiagnosticErrorV2{{
				File:    entryPoint,
				Message: fmt.Sprintf("Entry point not found in provided files: %s", entryPoint),
			}},
		}
	}

	// Get esbuild options from package.json
	opts := pkg.esbuildOptions()

	// Build resolver (same pattern as v2)
	resolverCalls := 0
	resolver := func(path string) (api.OnLoadResult, error) {
		resolverCalls++
		trackPackageResolution(path)

		if strings.HasPrefix(path, "/") || strings.HasPrefix(path, "./") || strings.HasPrefix(path, "../") {
			if content, exists := fs.ReadFile(path); exists {
				return api.OnLoadResult{
					Contents: &content,
					Loader:   loaderForPath(path),
				}, nil
			}
		}

		// Try with common extensions
		if strings.HasPrefix(path, "/") {
			extensions := []string{".js", ".jsx", ".mjs", ".json", ".ts", ".tsx"}
			for _, ext := range extensions {
				testPath := path + ext
				if content, exists := fs.ReadFile(testPath); exists {
					return api.OnLoadResult{
						Contents: &content,
						Loader:   loaderForPath(testPath),
					}, nil
				}
			}
		}

		// Bare module resolution
		if !strings.HasPrefix(path, "/") {
			resolvedPath := resolveModule(fs, path, "")
			if resolvedPath != "" {
				if content, exists := fs.ReadFile(resolvedPath); exists {
					return api.OnLoadResult{
						Contents: &content,
						Loader:   loaderForPath(resolvedPath),
					}, nil
				}
			}
		}

		return api.OnLoadResult{}, fmt.Errorf("file not found: %s", path)
	}

	// Configure esbuild
	opts.EntryPoints = []string{entryPoint}
	opts.Plugins = []api.Plugin{{
		Name: "virtual-fs-v3",
		Setup: func(pb api.PluginBuild) {
			pb.OnResolve(api.OnResolveOptions{Filter: ".*"}, func(args api.OnResolveArgs) (api.OnResolveResult, error) {
				trackPackageResolution(args.Path)

				// Handle absolute imports
				if strings.HasPrefix(args.Path, "/") {
					return api.OnResolveResult{Path: args.Path, Namespace: "virtual"}, nil
				}

				// Handle relative imports
				if strings.HasPrefix(args.Path, "./") || strings.HasPrefix(args.Path, "../") {
					resolvedPath := resolveModule(fs, args.Path, args.Importer)
					if resolvedPath == "" {
						importerPath := args.Importer
						if !strings.HasPrefix(importerPath, "/") {
							importerPath = resolveBarePackageImporter(fs, importerPath)
						}
						importerDir := filepath.Dir(importerPath)
						resolvedPath = filepath.Join(importerDir, args.Path)
						resolvedPath = strings.ReplaceAll(resolvedPath, "\\", "/")
					}
					return api.OnResolveResult{Path: resolvedPath, Namespace: "virtual"}, nil
				}

				// Handle bare imports
				if !strings.Contains(args.Path, "/") || strings.HasPrefix(args.Path, "@") {
					return api.OnResolveResult{Path: args.Path, Namespace: "virtual"}, nil
				}

				// Handle subpath imports from within node_modules
				if args.Importer != "" && strings.Contains(args.Importer, "/node_modules/") {
					parts := strings.Split(args.Importer, "/node_modules/")
					if len(parts) >= 2 {
						remainingPath := parts[1]
						packageParts := strings.Split(remainingPath, "/")
						packageName := packageParts[0]
						if strings.HasPrefix(packageName, "@") && len(packageParts) > 1 {
							packageName = packageParts[0] + "/" + packageParts[1]
						}
						resolvedPath := "/node_modules/" + packageName + "/" + args.Path
						return api.OnResolveResult{Path: resolvedPath, Namespace: "virtual"}, nil
					}
				}

				return api.OnResolveResult{Path: args.Path, Namespace: "virtual"}, nil
			})

			pb.OnLoad(api.OnLoadOptions{Filter: ".*", Namespace: "virtual"}, func(args api.OnLoadArgs) (api.OnLoadResult, error) {
				return resolver(args.Path)
			})
		},
	}}

	esbuildStart := time.Now()
	result := api.Build(opts)
	log.Printf("[PERF] esbuild.Build V3: %v (resolver called %d times)", time.Since(esbuildStart), resolverCalls)

	if len(result.Errors) > 0 {
		errors := make([]DiagnosticErrorV2, 0, len(result.Errors))
		for _, err := range result.Errors {
			diagErr := DiagnosticErrorV2{
				Message: err.Text,
			}
			if err.Location != nil {
				diagErr.File = err.Location.File
				diagErr.Line = err.Location.Line
				diagErr.Column = err.Location.Column
			}
			errors = append(errors, diagErr)
		}
		compileResults.WithLabelValues("error").Inc()
		return BuildV2Response{Errors: errors}
	}

	if len(result.OutputFiles) == 0 {
		compileResults.WithLabelValues("error").Inc()
		return BuildV2Response{Errors: []DiagnosticErrorV2{{Message: "No output generated"}}}
	}

	compileResults.WithLabelValues("success").Inc()
	return BuildV2Response{Code: string(result.OutputFiles[0].Contents)}
}

// loaderForPath returns the appropriate esbuild loader for a file path.
func loaderForPath(path string) api.Loader {
	switch {
	case strings.HasSuffix(path, ".tsx"), strings.HasSuffix(path, ".ts"):
		return api.LoaderTSX
	case strings.HasSuffix(path, ".jsx"):
		return api.LoaderJSX
	case strings.HasSuffix(path, ".json"):
		return api.LoaderJSON
	case strings.HasSuffix(path, ".mjs"):
		return api.LoaderJS
	default:
		return api.LoaderDefault
	}
}
```

```go
func getVersionsByModTime(cachePath string) ([]versionInfo, error) {
	entries, err := os.ReadDir(cachePath)
	if err != nil {
		return nil, err
	}

	var versions []versionInfo
	for _, entry := range entries {
		if !entry.IsDir() {
			continue
		}

		// If this is the deps directory, walk its children instead
		if entry.Name() == "deps" {
			depsPath := filepath.Join(cachePath, "deps")
			depEntries, err := os.ReadDir(depsPath)
			if err != nil {
				continue
			}
			for _, depEntry := range depEntries {
				if !depEntry.IsDir() {
					continue
				}
				info, err := depEntry.Info()
				if err != nil {
					continue
				}
				versions = append(versions, versionInfo{
					modTime: info.ModTime(),
					name:    filepath.Join("deps", depEntry.Name()),
				})
			}
			continue
		}

		info, err := entry.Info()
		if err != nil {
			continue
		}

		versions = append(versions, versionInfo{
			modTime: info.ModTime(),
			name:    entry.Name(),
		})
	}

	sort.Slice(versions, func(i, j int) bool {
		return versions[i].modTime.Before(versions[j].modTime)
	})

	return versions, nil
}
```
